C*
D A s 3p 0 inz(1)
D B s 3p 1 inz(10.0)
D C s 2p 0 inz(32)
D D s 2p 0 inz(-10)
D E s 3p 0 inz(6)
D F s 3p 0 inz(10)
D G s 3p 2 inz(2.77)
D H s 3p 0 inz(70)
D J s 3p 1 inz(0.6)
D K s 2p 0 inz(25)
D L s 2p 1 dim(3)
D V s 5p 2
D W s 5p 1
D X s 8p 4
D Y s 6p 2
D Z s 5p 3

/FREE
L(1) = 1.0;
L(2) = 1.7;
L(3) = -1.1;

A = A + 1; // A = 002
V = B + C; // V = 042.00
V = B + D; // V = 0
V = C; // V = 032.00
E = E - 1; // E = 005
W = C - B; // W = 0022.0
W = C - D; // W = 0042.0
W = - C; // W = -0032.0
F = F * E; // F = 060
X = B * G; // X = 0027.7000
X = B * D; // X = -0100.0000
H = H / B; // H = 007
Y = C / J; // Y = 0053.33
eval(r) Z = %sqrt(K); // Z = 05.000
Z = %xfoot(L); // Z = 01.600

dump(a);
*inlr = *on;
/END-FREE
